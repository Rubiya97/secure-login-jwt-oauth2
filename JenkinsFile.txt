pipeline {
    agent any

    environment {
        MAVEN_HOME = tool name: 'Maven 3.9.2', type: 'maven'  // Use your Jenkins Maven tool
        NODE_HOME = tool name: 'NodeJS 20', type: 'nodejs'    // Optional, if using Node.js plugin
        PATH = "${MAVEN_HOME}/bin:${NODE_HOME}/bin:${env.PATH}"
    }

    stages {
        stage('Build Backend') {
            steps {
                dir('oauth2') {
                    echo "Building backend..."
                    sh 'mvn clean package'
                }
            }
        }

        stage('Test Backend') {
            steps {
                dir('oauth2') {
                    echo "Running backend tests..."
                    sh 'mvn test'
                }
            }
        }

        stage('Build Frontend') {
            steps {
                dir('frontend-oauth') {
                    echo "Building frontend..."
                    sh 'npm install'
                    sh 'npm run build'
                }
            }
        }

        stage('Deploy') {
            steps {
                echo "Deployment step goes here (optional)"
            }
        }
    }

    post {
        success {
            echo "Pipeline completed successfully!"
        }
        failure {
            echo "Pipeline failed. Check logs for details."
        }
    }
}
